  (function initForensicExercise() {
    const btnScam = document.getElementById('btn-scam');
    const btnSafe = document.getElementById('btn-safe');
    const feedbackPanel = document.getElementById('feedback-panel');
    const targetLink = document.getElementById('target-link');
    const nextBtn = document.getElementById('next-case-btn');

    let audioEnabled = true;

    // Synthesize Bell/Chime audio using Web Audio API
    function playVictoryChime() {
      if (!audioEnabled) return;
      try {
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        if (!AudioContext) return;
        const ctx = new AudioContext();
        
        // Primary Bell Tone
        const osc1 = ctx.createOscillator();
        const gain1 = ctx.createGain();
        osc1.type = 'sine';
        osc1.frequency.setValueAtTime(880, ctx.currentTime); // A5
        osc1.frequency.exponentialRampToValueAtTime(440, ctx.currentTime + 0.8);
        gain1.gain.setValueAtTime(0.3, ctx.currentTime);
        gain1.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.8);
        osc1.connect(gain1);
        gain1.connect(ctx.destination);
        
        // Harmonics
        const osc2 = ctx.createOscillator();
        const gain2 = ctx.createGain();
        osc2.type = 'triangle';
        osc2.frequency.setValueAtTime(1760, ctx.currentTime); // A6
        gain2.gain.setValueAtTime(0.15, ctx.currentTime);
        gain2.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.5);
        osc2.connect(gain2);
        gain2.connect(ctx.destination);

        osc1.start();
        osc2.start();
        osc1.stop(ctx.currentTime + 0.8);
        osc2.stop(ctx.currentTime + 0.5);
      } catch (e) {
        // Fallback
      }
    }

    function triggerSuccess() {
      playVictoryChime();

      // Highlight clue inside chat bubble
      targetLink.classList.remove('bg-[#EAF3FF]', 'text-[#042C6C]', 'border-[#3BA2F7]/40');
      targetLink.classList.add('bg-rose-100', 'text-[#900B3B]', 'border-rose-300', 'font-bold');

      // Disable Buttons
      btnScam.disabled = true;
      btnSafe.disabled = true;
      btnScam.classList.add('opacity-50', 'pointer-events-none');
      btnSafe.classList.add('opacity-50', 'pointer-events-none');

      // Animate In Feedback Panel
      feedbackPanel.classList.remove('hidden');
      requestAnimationFrame(() => {
        feedbackPanel.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
      });
    }

    btnScam.addEventListener('click', triggerSuccess);

    // Micro-interaction on wrong choice

    });
